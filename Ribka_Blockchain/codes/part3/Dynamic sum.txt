pragma solidity 0.8.20;

contract Contract {
    // Pure external function that sums a dynamic array of unsigned integers
    function sum(uint256[] memory numbers) external pure returns (uint256) {
        uint256 total = 0;
        for (uint256 i = 0; i < numbers.length; i++) {
            total += numbers[i];
        }
        return total;
    }
}
